cmake_minimum_required(VERSION 3.24)
project(sdl_pdf_viewer VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(SDL_PDF_VIEWER_CORE_ONLY "Build PDF renderer and tests without SDL" OFF)
find_package(ZLIB REQUIRED)
find_package(JPEG REQUIRED)
find_package(PNG REQUIRED)
find_package(GIF REQUIRED)
add_library(pdf_core src/pdf.c src/raster.c src/cff.c src/type1.c src/brush.c src/animation.c src/text.c)
target_include_directories(pdf_core PUBLIC src)
target_link_libraries(pdf_core PUBLIC ZLIB::ZLIB JPEG::JPEG PNG::PNG GIF::GIF)
if(NOT MSVC)
  target_compile_options(pdf_core PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(pdf_core PUBLIC m)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_ink tests/test_ink.c)
  target_link_libraries(test_ink PRIVATE pdf_core)
  add_test(NAME ink COMMAND test_ink)
  add_executable(test_text tests/test_text.c)
  target_link_libraries(test_text PRIVATE pdf_core)
  add_test(NAME text COMMAND test_text)
  add_executable(test_raster tests/test_raster.c)
  target_link_libraries(test_raster PRIVATE pdf_core)
  add_test(NAME raster COMMAND test_raster)
  add_executable(test_animation tests/test_animation.c)
  target_link_libraries(test_animation PRIVATE pdf_core)
  add_test(NAME animation COMMAND test_animation "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_executable(benchmark_render tests/benchmark_render.c)
  target_link_libraries(benchmark_render PRIVATE pdf_core)
  add_executable(check_document tests/check_document.c)
  target_link_libraries(check_document PRIVATE pdf_core)
  add_test(NAME document COMMAND check_document "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/graphics.pdf")
  set(SDL_PDF_VIEWER_ACCEPTANCE_PDF "" CACHE FILEPATH "Optional PDF for full-document acceptance tests")
  if(SDL_PDF_VIEWER_ACCEPTANCE_PDF)
    add_test(NAME acceptance_pdf COMMAND check_document "${SDL_PDF_VIEWER_ACCEPTANCE_PDF}")
  endif()
  add_executable(check_links tests/check_links.c)
  target_link_libraries(check_links PRIVATE pdf_core)
  add_executable(render_document tests/render_document.c)
  target_link_libraries(render_document PRIVATE pdf_core)
  add_executable(test_pdf tests/test_pdf.c)
  target_link_libraries(test_pdf PRIVATE pdf_core)
  add_test(NAME pdf COMMAND test_pdf "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
endif()
if(SDL_PDF_VIEWER_CORE_ONLY)
  return()
endif()
# Prefer an installed SDL3 package (vcpkg, Homebrew, distro). Without one,
# fall back to building the pinned SDL release statically from source.
find_package(SDL3 3.4 CONFIG QUIET)
if(SDL3_FOUND)
  message(STATUS "Using installed SDL3 ${SDL3_VERSION}")
else()
  message(STATUS "SDL3 package not found; building SDL 3.4.20 from source")
  include(FetchContent)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL3
    URL https://www.libsdl.org/release/SDL3-3.4.20.tar.gz
    URL_HASH SHA256=f3bb06f55bed9557227a700af2ea62e49497ebdf58037732a731a5fb3ef8581f
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(SDL3)
endif()
add_executable(sdl-pdf-viewer src/main.c)
target_link_libraries(sdl-pdf-viewer PRIVATE pdf_core SDL3::SDL3)
if(NOT MSVC)
  target_compile_options(sdl-pdf-viewer PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS sdl-pdf-viewer RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_executable(test_app tests/test_app.c)
  target_link_libraries(test_app PRIVATE pdf_core SDL3::SDL3)
  add_test(NAME app COMMAND test_app "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  set_tests_properties(app PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software")
  add_executable(test_animation_app tests/test_animation_app.c)
  target_link_libraries(test_animation_app PRIVATE pdf_core SDL3::SDL3)
  add_test(NAME animation_app COMMAND test_animation_app "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  set_tests_properties(animation_app PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software")
endif()
